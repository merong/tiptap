플러그인 시스템

Tiptap의 확장 시스템을 활용한 플러그인 예제입니다.

활성화된 확장 기능

커스텀 Extension 예제

import { Extension } from '@tiptap/core'

const CharacterCount = Extension.create({
  name: 'characterCount',

  addStorage() {
    return {
      characters: 0,
      words: 0
    }
  },

  onUpdate() {
    const text = this.editor.getText()
    this.storage.characters = text.length
    this.storage.words = text.split(/\s+/).filter(w => w).length
  }
})

// 사용
const editor = new Editor({
  extensions: [StarterKit, CharacterCount],
})

console.log(editor.storage.characterCount.words)
