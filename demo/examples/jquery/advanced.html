jQuery 이벤트 통합

jQuery의 이벤트 시스템과 Tiptap을 통합한 고급 예제입니다.

실시간 이벤트 모니터

jQuery 이벤트 예제

// 커스텀 이벤트 리스너
$('#editor')
  .on('tiptap:initialized', function(e, instance) {
    console.log('Editor initialized')
  })
  .on('tiptap:update', function(e, editor) {
    console.log('Content updated')
  })
  .on('tiptap:saved', function(e, item) {
    console.log('Content saved:', item)
  })
  .on('tiptap:loaded', function(e, item) {
    console.log('Content loaded:', item)
  })

// jQuery 애니메이션과 함께 사용
$('#editor')
  .fadeOut()
  .setEditorContent('<p>New content</p>')
  .fadeIn()
