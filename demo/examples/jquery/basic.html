jQuery 플러그인 패턴

jQuery 플러그인 패턴을 사용한 Tiptap 에디터 구현입니다.

코드 예제

// jQuery 플러그인으로 에디터 초기화
$('#editor').tiptapEditor({
  placeholder: '여기에 입력하세요...',
  content: '<p>안녕하세요!</p>',
  onChange: function(editor) {
    console.log('Content changed')
  }
})

// 메서드 호출
const instance = $('#editor').data('tiptapEditor')
const html = instance.getHTML()
instance.setContent('<p>New content</p>')

// jQuery 이벤트 리스너
$('#editor').on('tiptap:update', function(e, editor) {
  console.log('Updated')
})

$('#editor').on('tiptap:saved', function(e, item) {
  console.log('Saved:', item)
})
