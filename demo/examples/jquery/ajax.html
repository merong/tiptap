AJAX를 통한 콘텐츠 로딩

jQuery AJAX를 사용하여 서버에서 콘텐츠를 불러오고 저장하는 예제입니다.

이 예제는 데모 목적으로 localStorage를 사용하여 AJAX 동작을 시뮬레이션합니다.

AJAX 코드 예제

// 문서 목록 로드
$.ajax({
  url: '/api/documents',
  method: 'GET'
})
.done(function(documents) {
  documents.forEach(function(doc) {
    // 목록 표시
  })
})

// 문서 불러오기
$.ajax({
  url: '/api/documents/' + id,
  method: 'GET'
})
.done(function(document) {
  $('#editor').data('editor').commands.setContent(document.content)
})

// 문서 저장
$.ajax({
  url: '/api/documents',
  method: 'POST',
  contentType: 'application/json',
  data: JSON.stringify({
    title: 'Document Title',
    content: $('#editor').data('editor').getHTML()
  })
})
.done(function(response) {
  console.log('Saved:', response)
})
